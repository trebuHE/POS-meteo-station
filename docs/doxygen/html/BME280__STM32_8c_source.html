62int BME280_Config (uint8_t osrs_t, uint8_t osrs_p, uint8_t osrs_h, uint8_t mode, uint8_t t_sb, uint8_t filter)
63{
64 // Read the Trimming parameters
65 if (TrimRead() != 0)
66 {
67 return 1;
68 }
69
70 uint8_t datatowrite = 0;
71 uint8_t datacheck = 0;
72
73 // Reset the device
74 datatowrite = 0xB6; // reset sequence
75 if (HAL_I2C_Mem_Write(&BME280_I2C, BME280_ADDRESS, RESET_REG, 1, &datatowrite, 1, 1000) != HAL_OK)
76 {
77 return 2; //error NUM for debug
78 }
79
80 HAL_Delay (100);
81
82
83 // write the humidity oversampling to 0xF2
84 datatowrite = osrs_h;
85 if (HAL_I2C_Mem_Write(&BME280_I2C, BME280_ADDRESS, CTRL_HUM_REG, 1, &datatowrite, 1, 1000) != HAL_OK)
86 {
87 return 3; //error NUM for debug
88 }
89 HAL_Delay (100);
90 HAL_I2C_Mem_Read(&BME280_I2C, BME280_ADDRESS, CTRL_HUM_REG, 1, &datacheck, 1, 1000);
91 if (datacheck != datatowrite)
92 {
93 return 4; //error NUM for debug
94 }
95
96
97 // write the standby time and IIR filter coeff to 0xF5
98 datatowrite = (t_sb <<5) |(filter << 2);
99 if (HAL_I2C_Mem_Write(&BME280_I2C, BME280_ADDRESS, CONFIG_REG, 1, &datatowrite, 1, 1000) != HAL_OK)
100 {
101 return 5; //error NUM for debug
102 }
103 HAL_Delay (100);
104 HAL_I2C_Mem_Read(&BME280_I2C, BME280_ADDRESS, CONFIG_REG, 1, &datacheck, 1, 1000);
105 if (datacheck != datatowrite)
106 {
107 return 6; //error NUM for debug
108 }
109
110
111 // write the pressure and temp oversampling along with mode to 0xF4
112 datatowrite = (osrs_t <<5) |(osrs_p << 2) | mode;
113 if (HAL_I2C_Mem_Write(&BME280_I2C, BME280_ADDRESS, CTRL_MEAS_REG, 1, &datatowrite, 1, 1000) != HAL_OK)
114 {
115 return 7; //error NUM for debug
116 }
117 HAL_Delay (100);
118 HAL_I2C_Mem_Read(&BME280_I2C, BME280_ADDRESS, CTRL_MEAS_REG, 1, &datacheck, 1, 1000);
119 if ((datacheck >> 2) != (datatowrite >> 2)) //ignore mode setting, forced mode goes instantly to sleep mode throwing the check off
120 {
121 return 8; //error NUM for debug
122 }
123
124 return 0; //Success
125}

152void BME280_Measure(float *temperature, float *pressure, float *humidity)
153{
154 if (BMEReadRaw() == 0)
155 {
156 if (tRaw == 0x800000) *temperature = 0; // temp disabled
157 else
158 *temperature = (BME280_compensate_T_int32(tRaw)) / 100.0f;
159
160 if (pRaw == 0x800000) *pressure = 0; // pressure disabled
161 else
162 {
163#if SUPPORT_64BIT
164 *pressure = (BME280_compensate_P_int64(pRaw)) / 256.0f;
165#elif SUPPORT_32BIT
166 *pressure = (BME280_compensate_P_int32(pRaw)); // in Pa
167#endif
168 }
169
170 if (hRaw == 0x8000) *humidity = 0; // humidity disabled
171 else
172 *humidity = (bme280_compensate_H_int32(hRaw)) / 1024.0f;
173 }
174 else
175 {
176 // if the device is detached
177 *temperature = 0;
178 *pressure = 0;
179 *humidity = 0;
180 }
181}

188static int TrimRead(void)
189{
190 uint8_t trimdata[32];
191
192 // Read NVM from 0x88 to 0xA1 (25 bytes)
193 if (HAL_I2C_Mem_Read(&BME280_I2C, BME280_ADDRESS, 0x88, 1, trimdata, 25, HAL_MAX_DELAY) != HAL_OK)
194 {
195 return 1; //error NUM for debug
196 }
197
198 // Read NVM from 0xE1 to 0xE7 (7 bytes)
199 if (HAL_I2C_Mem_Read(&BME280_I2C, BME280_ADDRESS, 0xE1, 1, trimdata + 25, 7, HAL_MAX_DELAY) != HAL_OK)
200 {
201 return 2; //error NUM for debug
202 }
203
204 // Temperature coefficients
205 dig_T1 = (uint16_t)(trimdata[1] << 8 | trimdata[0]);
206 dig_T2 = (int16_t)(trimdata[3] << 8 | trimdata[2]);
207 dig_T3 = (int16_t)(trimdata[5] << 8 | trimdata[4]);
208
209 // Pressure coefficients
210 dig_P1 = (uint16_t)(trimdata[7] << 8 | trimdata[6]);
211 dig_P2 = (int16_t)(trimdata[9] << 8 | trimdata[8]);
212 dig_P3 = (int16_t)(trimdata[11] << 8 | trimdata[10]);
213 dig_P4 = (int16_t)(trimdata[13] << 8 | trimdata[12]);
214 dig_P5 = (int16_t)(trimdata[15] << 8 | trimdata[14]);
215 dig_P6 = (int16_t)(trimdata[17] << 8 | trimdata[16]);
216 dig_P7 = (int16_t)(trimdata[19] << 8 | trimdata[18]);
217 dig_P8 = (int16_t)(trimdata[21] << 8 | trimdata[20]);
218 dig_P9 = (int16_t)(trimdata[23] << 8 | trimdata[22]);
219
220 // Humidity coefficients
221 dig_H1 = trimdata[24];
222 dig_H2 = (int16_t)(trimdata[26] << 8 | trimdata[25]);
223 dig_H3 = trimdata[27];
224 dig_H4 = (int16_t)((trimdata[28] << 4) | (trimdata[29] & 0x0F));
225 dig_H5 = (int16_t)((trimdata[30] << 4) | (trimdata[29] >> 4));
226 dig_H6 = (int8_t)trimdata[31];
227
228 return 0; //Success
229}

235static int BMEReadRaw(void)
236{
237 uint8_t RawData[8];
238
239 // Check the chip ID before reading
240 HAL_I2C_Mem_Read(&hi2c1, BME280_ADDRESS, ID_REG, 1, &chipID, 1, 1000);
241
242 if (chipID == 0x60)
243 {
244 // Read the Registers 0xF7 to 0xFE
245 HAL_I2C_Mem_Read(&BME280_I2C, BME280_ADDRESS, PRESS_MSB_REG, 1, RawData, 8, HAL_MAX_DELAY);
246
247 /* Calculate the Raw data for the parameters
248 * Here the Pressure and Temperature are in 20 bit format and humidity in 16 bit format
249 */
250 pRaw = (RawData[0]<<12)|(RawData[1]<<4)|(RawData[2]>>4);
251 tRaw = (RawData[3]<<12)|(RawData[4]<<4)|(RawData[5]>>4);
252 hRaw = (RawData[6]<<8)|(RawData[7]);
253
254 return 0;
255 }
256
257 else return 1; //error NUM for debug
258}
