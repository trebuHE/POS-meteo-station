67void CC1101_Config(void) {
68 // 1. Reset Chip
69 CC1101_CommandStrobe(CC1101_SRES);
70
71 // 2. IO Config
72 // GDO0: Asserts on Sync Word, De-asserts on End of Packet
73 CC1101_WriteReg(CC1101_IOCFG0, 0x06);
74
75 // 3. FIFO & Packet Control
76 CC1101_WriteReg(CC1101_FIFOTHR, 0x47); // Threshold: 32 bytes
77 CC1101_WriteReg(CC1101_ADDR, 0x10); // Device Address: 0x10
78 CC1101_WriteReg(CC1101_PKTCTRL0, 0x05); // Variable Length, CRC Enabled
79 // Append Status (RSSI/LQI) + Address Check (0x00 is Broadcast)
80 CC1101_WriteReg(CC1101_PKTCTRL1, 0x06);
81
82 // 4. Frequency (433.92 MHz)
83 CC1101_WriteReg(CC1101_FREQ2, 0x10);
84 CC1101_WriteReg(CC1101_FREQ1, 0xB1);
85 CC1101_WriteReg(CC1101_FREQ0, 0x3B);
86
87 // 5. Modem Config (1.2 kBaud, GFSK, ~58kHz RX BW)
88 CC1101_WriteReg(CC1101_MDMCFG4, 0xF5);
89 CC1101_WriteReg(CC1101_MDMCFG3, 0x83);
90 CC1101_WriteReg(CC1101_MDMCFG2, 0x11); // 15/16 Sync Bits detected
91 CC1101_WriteReg(CC1101_DEVIATN, 0x15);
92
93 // 6. State Machine (MCSM)
94 // Calibrate on IDLE->RX/TX transition
95 CC1101_WriteReg(CC1101_MCSM0, 0x18);
96 // Stay in RX after receiving a packet (Critical for RX mode)
97 CC1101_WriteReg(CC1101_MCSM1, 0x0F);
98
99 // 7. Power Settings (0 dBm)
100 CC1101_WriteReg(CC1101_FOCCFG, 0x16);
101 CC1101_WriteReg(CC1101_WORCTRL, 0xFB);
102
103 uint8_t paTable[] = {0x60, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00};
104 CC1101_WriteBurst(CC1101_PATABLE, paTable, 8);
105}

133void CC1101_Transmit(uint8_t *data, uint8_t length) {
134 CC1101_CommandStrobe(CC1101_SIDLE);
135 CC1101_CommandStrobe(CC1101_SFTX); // Flush TX buffer
136

150uint8_t CC1101_Receive(uint8_t *data) {
151 uint8_t rxBytes = CC1101_ReadStatus(CC1101_RXBYTES) & 0x7F;
152
153 // Check if we have at least the Length byte
154 if (rxBytes > 1) {
155 uint8_t len = CC1101_ReadReg(CC1101_FIFO);
156
157 // Sanity Check: If length is unreasonably large, it's noise
158 if (len > 60) {
159 CC1101_CommandStrobe(CC1101_SIDLE);
160 CC1101_CommandStrobe(CC1101_SFRX);
161 CC1101_SetRx();
162 return 0;
163 }
164
165 // Wait for the full packet to arrive (Payload + 2 Status Bytes)
166 uint8_t bytesNeeded = len + 2;
167 uint32_t timeout = HAL_GetTick() + 500;
168
169 while (1) {
170 rxBytes = CC1101_ReadStatus(CC1101_RXBYTES) & 0x7F;
171 if (rxBytes >= bytesNeeded) break;
172
173 if (HAL_GetTick() > timeout) {
174 CC1101_CommandStrobe(CC1101_SIDLE);
175 CC1101_CommandStrobe(CC1101_SFRX);
176 CC1101_SetRx();
177 return 0;
178 }
179 HAL_Delay(2);
180 }
181
182 // Read Payload
183 CC1101_ReadBurst(CC1101_FIFO, data, len);
184
185 // Read & Check Status Bytes (RSSI + CRC_OK)
186 uint8_t status[2];
187 CC1101_ReadBurst(CC1101_FIFO, status, 2);
188
189 // CRC Check (Bit 7 of 2nd status byte)
190 if (!(status[1] & 0x80)) {
191 CC1101_SetRx(); // CRC Failed, ignore packet
192 return 0;
193 }
194
195 CC1101_SetRx(); // Return to listening
196 return len;
197 }
198 return 0;
199}
