105void CC1101_Config(void) {
106 // 1. Reset Chip
107 CC1101_CommandStrobe(CC1101_SRES);
108
109 // 2. IO Config
110 // GDO0: Asserts on Sync Word, De-asserts on End of Packet
111 CC1101_WriteReg(CC1101_IOCFG0, 0x06);
112
113 // 3. FIFO & Packet Control
114 CC1101_WriteReg(CC1101_FIFOTHR, 0x47); // Threshold: 32 bytes
115 CC1101_WriteReg(CC1101_ADDR, 0x10); // Device Address: 0x10
116 CC1101_WriteReg(CC1101_PKTCTRL0, 0x05); // Variable Length, CRC Enabled
117 // Append Status (RSSI/LQI) + Address Check (0x00 is Broadcast)
118 CC1101_WriteReg(CC1101_PKTCTRL1, 0x06);
119
120 // 4. Frequency (433.92 MHz)
121 CC1101_WriteReg(CC1101_FREQ2, 0x10);
122 CC1101_WriteReg(CC1101_FREQ1, 0xB1);
123 CC1101_WriteReg(CC1101_FREQ0, 0x3B);
124
125 // 5. Modem Config (1.2 kBaud, GFSK, ~58kHz RX BW)
126 CC1101_WriteReg(CC1101_MDMCFG4, 0xF5);
127 CC1101_WriteReg(CC1101_MDMCFG3, 0x83);
128 CC1101_WriteReg(CC1101_MDMCFG2, 0x11); // 15/16 Sync Bits detected
129 CC1101_WriteReg(CC1101_DEVIATN, 0x15);
130
131 // 6. State Machine (MCSM)
132 // Calibrate on IDLE->RX/TX transition
133 CC1101_WriteReg(CC1101_MCSM0, 0x18);
134 // Stay in RX after receiving a packet (Critical for RX mode)
135 CC1101_WriteReg(CC1101_MCSM1, 0x0F);
136
137 // 7. Power Settings (0 dBm)
138 CC1101_WriteReg(CC1101_FOCCFG, 0x16);
139 CC1101_WriteReg(CC1101_WORCTRL, 0xFB);
140
141 uint8_t paTable[] = {0x60, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00};
142 CC1101_WriteBurst(CC1101_PATABLE, paTable, 8);
143}

174void CC1101_Transmit(uint8_t *data, uint8_t length) {
175 CC1101_CommandStrobe(CC1101_SIDLE);
176 CC1101_CommandStrobe(CC1101_SFTX); // Flush TX buffer
177
178 // Format: [Length] [Data...]
179 CC1101_WriteReg(CC1101_FIFO, length);
180 CC1101_WriteBurst(CC1101_FIFO, data, length);
181
182 CC1101_CommandStrobe(CC1101_STX);
183
184 // Wait for Sync Word (GDO0 High) then End of Packet (GDO0 Low)
185 while (HAL_GPIO_ReadPin(CC1101_GDO0_PORT, CC1101_GDO0_PIN) == GPIO_PIN_RESET);
186 while (HAL_GPIO_ReadPin(CC1101_GDO0_PORT, CC1101_GDO0_PIN) == GPIO_PIN_SET);
187
188 CC1101_CommandStrobe(CC1101_SIDLE);
189}

192uint8_t CC1101_Receive(uint8_t *data) {
193 uint8_t rxBytes = CC1101_ReadStatus(CC1101_RXBYTES) & 0x7F;
194
195 // Check if we have at least the Length byte
196 if (rxBytes > 1) {
197 uint8_t len = CC1101_ReadReg(CC1101_FIFO);
198
199 // Sanity Check: If length is unreasonably large, it's noise
200 if (len > 60) {
201 CC1101_CommandStrobe(CC1101_SIDLE);
202 CC1101_CommandStrobe(CC1101_SFRX);
203 CC1101_SetRx();
204 return 0;
205 }
206
207 // Wait for the full packet to arrive (Payload + 2 Status Bytes)
208 uint8_t bytesNeeded = len + 2;
209 uint32_t timeout = HAL_GetTick() + 500;
210
211 while (1) {
212 rxBytes = CC1101_ReadStatus(CC1101_RXBYTES) & 0x7F;
213 if (rxBytes >= bytesNeeded) break;
214
215 if (HAL_GetTick() > timeout) {
216 CC1101_CommandStrobe(CC1101_SIDLE);
217 CC1101_CommandStrobe(CC1101_SFRX);
218 CC1101_SetRx();
219 return 0;
220 }
221 HAL_Delay(2);
222 }
223
224 // Read Payload
225 CC1101_ReadBurst(CC1101_FIFO, data, len);
226
227 // Read & Check Status Bytes (RSSI + CRC_OK)
228 uint8_t status[2];
229 CC1101_ReadBurst(CC1101_FIFO, status, 2);
230
231 // CRC Check (Bit 7 of 2nd status byte)
232 if (!(status[1] & 0x80)) {
233 CC1101_SetRx(); // CRC Failed, ignore packet
234 return 0;
235 }
236
237 CC1101_SetRx(); // Return to listening
238 return len;
239 }
240 return 0;
241}
