56void GPS_Append_NMEA_Checksum(char *str, size_t size) {
57 if (!str || size < 6) return;
58
59 size_t len = strlen(str);
60 if (len == 0 || len + 5 >= size) return; // Not enough space
61
62 // Compute checksum: XOR of all chars between '$' and '*'
63 uint8_t checksum = 0;
64 const char *ptr = str;
65
66 if (*ptr == '$') ptr++; // skip initial $
67
68 while (*ptr && *ptr != '*')
69 checksum ^= (uint8_t)*ptr++;
70
71 int written = snprintf(str + len, size - len, "*%02X\r\n", checksum);
72
73 if (written < 0 || (size_t)written >= size - len)
74 str[size - 1] = '\0';
75}

120int GPS_validate(char *nmeastr){
121 char check[3];
122 char checkcalcstr[3];
123 int i;
124 int calculated_check;
125
126 i=0;
127 calculated_check=0;
128
129 // check to ensure that the string starts with a $
130 if(nmeastr[i] == '$')
131 i++;
132 else
133 return 0;
134
135 //No NULL reached, 75 char largest possible NMEA message, no '*' reached
136 while((nmeastr[i] != 0) && (nmeastr[i] != '*') && (i < 75)){
137 calculated_check ^= nmeastr[i];// calculate the checksum
138 i++;
139 }
140
141 if(i >= 75){
142 return 0;// the string was too long so return an error
143 }
144
145 if (nmeastr[i] == '*'){
146 check[0] = nmeastr[i+1]; //put hex chars in check string
147 check[1] = nmeastr[i+2];
148 check[2] = 0;
149 }
150 else
151 return 0;// no checksum separator found there for invalid
152
153 sprintf(checkcalcstr,"%02X",calculated_check);
154 return((checkcalcstr[0] == check[0])
155 && (checkcalcstr[1] == check[1])) ? 1 : 0 ;
